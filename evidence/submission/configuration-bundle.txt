#!/bin/bash
set -euo pipefail
ROOT=$(cd "$(dirname "$0")/.." && pwd)
source "$ROOT/phase2/lab.env"
ROLE=${1:?Usage: bash phase2/render.sh mac1|mac2}
case "$ROLE" in
  mac1) OWN_IP=$MAC1_IP; A_ADDR=$MAC2_IP; B_ADDR=127.0.0.1; EDGE_ID=mac1 ;;
  mac2) OWN_IP=$MAC2_IP; A_ADDR=127.0.0.1; B_ADDR=$MAC1_IP; EDGE_ID=mac2 ;;
  *) echo 'Expected mac1 or mac2' >&2; exit 1 ;;
esac
RUNTIME="$ROOT/.runtime"
mkdir -p "$RUNTIME" "$ROOT/evidence/phase-2"/{baseline,dns-failover,ttl,ha,isolation,edge-migration,troubleshooting}
chmod 700 "$RUNTIME"
if [ ! -f "$RUNTIME/records.hosts" ]; then
  printf '%s app.cn-project.test api.cn-project.test\n' "$MAC2_IP" > "$RUNTIME/records.hosts"
fi
cat > "$RUNTIME/dnsmasq.conf" <<EOF
port=53
listen-address=127.0.0.1,$OWN_IP
bind-interfaces
no-resolv
no-hosts
local=/cn-project.test/
addn-hosts=$RUNTIME/records.hosts
local-ttl=$DNS_TTL
cache-size=1000
pid-file=$RUNTIME/dnsmasq.pid
log-queries
log-facility=$RUNTIME/dnsmasq.log
EOF
cat > "$RUNTIME/resolver.conf" <<EOF
nameserver $MAC1_IP
nameserver $MAC2_IP
EOF
cat > "$RUNTIME/nginx.conf" <<EOF
pid $RUNTIME/nginx.pid;
error_log $RUNTIME/nginx-error.log info;
events {}
http {
    log_format upstream_trace '\$time_iso8601 client=\$remote_addr edge=$EDGE_ID status=\$status upstream=\$upstream_addr upstream_status=\$upstream_status time=\$upstream_response_time';
    access_log $RUNTIME/nginx-access.log upstream_trace;
    upstream cn_backends {
        server $A_ADDR:3001 max_fails=1 fail_timeout=5s;
        server $B_ADDR:3002 max_fails=1 fail_timeout=5s;
    }
    server {
        listen 8080;
        listen 8443 ssl;
        server_name app.cn-project.test api.cn-project.test;
        ssl_certificate $RUNTIME/server.crt;
        ssl_certificate_key $RUNTIME/server.key;
        ssl_protocols TLSv1.2 TLSv1.3;
        add_header X-Edge $EDGE_ID always;
        location / {
            proxy_pass http://cn_backends;
            proxy_http_version 1.1;
            proxy_set_header Host \$host;
            proxy_set_header X-Forwarded-For \$proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto \$scheme;
            proxy_connect_timeout 2s;
            proxy_read_timeout 5s;
            proxy_next_upstream error timeout http_502 http_503 http_504;
            proxy_next_upstream_tries 2;
        }
    }
}
EOF
printf '%s\n' "$ROLE" > "$RUNTIME/role"
/opt/homebrew/sbin/dnsmasq --test --conf-file="$RUNTIME/dnsmasq.conf"
echo "Rendered $ROLE configs in $RUNTIME"


=== phase2/install-admin.sh ===
#!/bin/bash
# Run once on each Mac with its local administrator password.
set -euo pipefail
ROOT=$(cd "$(dirname "$0")/.." && pwd)
ROLE=${1:?Usage: bash phase2/install-admin.sh mac1|mac2}
case "$ROLE" in mac1|mac2) ;; *) exit 2 ;; esac
if [ "$EUID" -ne 0 ]; then
  exec sudo /bin/bash "$ROOT/phase2/install-admin.sh" "$ROLE"
fi
export PATH=/usr/bin:/bin:/usr/sbin:/sbin
ACCOUNT=${SUDO_USER:?Run this installer through sudo from your normal account}
[[ "$ACCOUNT" =~ ^[a-zA-Z_][a-zA-Z0-9_-]*$ ]] || exit 2
source "$ROOT/phase2/lab.env"
STATE=/private/var/db/cn-project
HELPER=/usr/local/sbin/cn-project-admin
RULE=/private/etc/sudoers.d/cn-project
if [ -e "$HELPER" ] || [ -e "$RULE" ]; then
  echo 'Project admin installation exists; refusing to overwrite it.' >&2
  exit 1
fi
install -d -o root -g wheel -m 755 /usr/local/sbin /private/etc/sudoers.d
install -d -o root -g wheel -m 755 "$STATE"
install -o root -g wheel -m 600 "$ROOT/phase2/lab.env" "$STATE/lab.env"
install -o root -g wheel -m 755 "$ROOT/phase2/admin-control.sh" "$HELPER"
OWN_IP=$MAC1_IP
PORT=3002
if [ "$ROLE" = mac2 ]; then OWN_IP=$MAC2_IP; PORT=3001; fi
printf '%s app.cn-project.test api.cn-project.test\n' "$MAC2_IP" > "$STATE/records.hosts"
chmod 644 "$STATE/records.hosts"
cat > "$STATE/dnsmasq.conf" <<EOF
port=53
listen-address=127.0.0.1,$OWN_IP
bind-interfaces
no-resolv
no-hosts
local=/cn-project.test/
addn-hosts=$STATE/records.hosts
local-ttl=$DNS_TTL
cache-size=1000
pid-file=$STATE/dnsmasq.pid
log-queries
log-facility=$STATE/dnsmasq.log
EOF
# Normal stage allows only the primary edge's LAN address. Roll back before cutover.
cat > "$STATE/isolation.pf" <<EOF
pass in quick on $LAB_INTERFACE inet proto tcp from $MAC2_IP to $OWN_IP port $PORT keep state
block drop in quick on $LAB_INTERFACE inet proto tcp from any to $OWN_IP port $PORT
EOF
mkdir -p /etc/resolver
if [ -f /etc/resolver/cn-project.test ]; then
  cp -p /etc/resolver/cn-project.test "$STATE/resolver.original"
fi
printf 'nameserver %s\nnameserver %s\n' "$MAC1_IP" "$MAC2_IP" > /etc/resolver/cn-project.test
chmod 644 /etc/resolver/cn-project.test
/opt/homebrew/sbin/dnsmasq --test --conf-file="$STATE/dnsmasq.conf"
# Exact actions only; the helper and all privileged configs are root-owned.
temp_rule=$(mktemp /private/tmp/cn-project-sudoers.XXXXXX)
trap 'rm -f "$temp_rule"' EXIT
printf 'Cmnd_Alias CN_PROJECT = ' > "$temp_rule"
separator=''
for action in dns-start dns-stop dns-primary dns-standby dns-status cache-flush firewall-on firewall-off firewall-status; do
  printf '%s%s %s' "$separator" "$HELPER" "$action" >> "$temp_rule"
  separator=', '
done
printf '\n%s ALL=(root) NOPASSWD: CN_PROJECT\n' "$ACCOUNT" >> "$temp_rule"
/usr/sbin/visudo -cf "$temp_rule"
install -o root -g wheel -m 440 "$temp_rule" "$RULE"
/usr/sbin/visudo -c
echo 'Installed limited password-free project controls.'
"$HELPER" dns-start
"$HELPER" cache-flush
echo 'Ready: sudo -n /usr/local/sbin/cn-project-admin dns-status'


=== phase2/admin-control.sh ===
#!/bin/bash
# Installed root-owned; accepts only the fixed project operations below.
set -euo pipefail
export PATH=/usr/bin:/bin:/usr/sbin:/sbin
unset BASH_ENV ENV CDPATH
STATE=/private/var/db/cn-project
source "$STATE/lab.env"
if [ "$EUID" -ne 0 ] || [ "$#" -ne 1 ]; then
  echo 'Use sudo cn-project-admin ACTION' >&2
  exit 2
fi
DNSMASQ=/opt/homebrew/sbin/dnsmasq
PF=/sbin/pfctl
ANCHOR=com.apple/cn-project
dns_pid() {
  test -f "$STATE/dnsmasq.pid" || return 1
  local pid
  pid=$(cat "$STATE/dnsmasq.pid")
  [[ "$pid" =~ ^[0-9]+$ ]] || return 1
  kill -0 "$pid" 2>/dev/null || return 1
  ps -p "$pid" -o comm= | grep -q 'dnsmasq$' || return 1
  printf '%s\n' "$pid"
}
case "$1" in
  dns-start)
    if dns_pid >/dev/null; then
      echo 'Project DNS already running'
    else
      "$DNSMASQ" --test --conf-file="$STATE/dnsmasq.conf"
      "$DNSMASQ" --conf-file="$STATE/dnsmasq.conf"
    fi
    ;;
  dns-stop)
    pid=$(dns_pid)
    kill -TERM "$pid"
    ;;
  dns-primary|dns-standby)
    target=$MAC2_IP
    [ "$1" != dns-standby ] || target=$MAC1_IP
    printf '%s app.cn-project.test api.cn-project.test\n' "$target" > "$STATE/records.hosts"
    if pid=$(dns_pid); then kill -HUP "$pid"; fi
    cat "$STATE/records.hosts"
    ;;
  dns-status)
    dns_pid || true
    cat "$STATE/records.hosts"
    tail -n 25 "$STATE/dnsmasq.log" 2>/dev/null || true
    ;;
  cache-flush)
    /usr/bin/dscacheutil -flushcache
    /usr/bin/killall -HUP mDNSResponder
    ;;
  firewall-on)
    # Use the existing Apple wildcard anchor; preserve the main PF ruleset.
    "$PF" -sr 2>/dev/null | grep -F 'anchor "com.apple/*"' >/dev/null || {
      echo 'Expected Apple anchor is missing; refusing to change the main ruleset.' >&2
      exit 1
    }
    "$PF" -a "$ANCHOR" -nf "$STATE/isolation.pf"
    if [ ! -f "$STATE/pf.token" ]; then
      enabled=$("$PF" -E 2>&1)
      printf '%s\n' "$enabled"
      printf '%s\n' "$enabled" | awk '/Token/ {print $NF}' > "$STATE/pf.token"
      test -s "$STATE/pf.token" || { rm -f "$STATE/pf.token"; exit 1; }
    fi
    "$PF" -a "$ANCHOR" -f "$STATE/isolation.pf"
    "$PF" -a "$ANCHOR" -vvsr
    ;;
  firewall-off)
    "$PF" -a "$ANCHOR" -F rules
    if [ -s "$STATE/pf.token" ]; then
      token=$(cat "$STATE/pf.token")
      [[ "$token" =~ ^[0-9]+$ ]] || exit 1
      "$PF" -X "$token"
      rm -f "$STATE/pf.token"
    fi
    ;;
  firewall-status)
    "$PF" -s info
    "$PF" -a "$ANCHOR" -vvsr
    ;;
  *) echo 'Allowed: dns-start dns-stop dns-primary dns-standby dns-status cache-flush firewall-on firewall-off firewall-status' >&2; exit 2 ;;
esac
